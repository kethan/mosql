  { _id: 3, name: 'Charlie', age: 22, city: 'Berlin', active: false, tags: [], profile: { country: 'Germany', score: 60 } },
  { _id: 4, name: 'David', age: 40, city: 'Paris', active: true, tags: [], profile: { country: 'France', score: 95 } },
